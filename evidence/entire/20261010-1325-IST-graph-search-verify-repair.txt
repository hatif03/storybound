def oath_hash(commitments: list[Commitment]) -> str:
    blob = json.dumps(
        [c.model_dump() for c in sorted(commitments, key=lambda c: c.id)], sort_keys=True, ensure_ascii=False
    )
    return hashlib.sha256(blob.encode("utf-8")).hexdigest()

def page_block(p: Page) -> str:
    lines = [f"PAGE {p.index}", f"Scenario: {p.scenario}", f"Question: {p.question}"]
    for c in p.choices:
        lines.append(f"Choice {c.id} [{'CORRECT' if c.is_correct else 'WRONG'}]: {c.text} | Feedback: {c.feedback}")
    lines.append(f"Image prompt: {p.image_prompt}")
    return "\n".join(lines)

def author(lesson: Lesson, card: CharacterCard) -> tuple[Oath, dict]:
    base = catalog.core_oath()
    info: dict = {"rejected": [], "added": [], "tokens_in": 0, "tokens_out": 0, "author_failed": False}
    try:
        r = ask_json(
            "oath_author",
            prompts.OATH_SYSTEM,
            prompts.OATH_USER.format(
                lesson=f"{lesson.title}: {lesson.concept}", card=json.dumps(card.model_dump(), ensure_ascii=False)
            ),
            temperature=0.3,
            max_tokens=900,
        )
        adds = r.value.get("additions", []) if isinstance(r.value, dict) else []
        oath, rejected = catalog.merge_additions(base, [a for a in adds if isinstance(a, dict)])
        info.update(
            rejected=rejected,
            added=[c.id for c in oath.commitments if not c.core],
            tokens_in=r.usage.input_tokens,
            tokens_out=r.usage.output_tokens,
        )
        return oath, info
    except ClientError:
        info["author_failed"] = True  # the core oath alone is still a complete oath
        return base, info
